\section{Research program}\label{sec:program}

Each branch states what is proved, what is open, and the first concrete step.

\begin{enumerate}[label=\arabic*.,leftmargin=*]

\item \textbf{Infinite-horizon averaging for McKean--Vlasov closed loops (Conjecture H.6).}
  \begin{itemize}
    \item \emph{Proved:} the averaged system converges globally and exponentially (H(ii)); the exact first correction
      at frozen measure (H.7); input-to-state stability of the averaged system (H.8).
    \item \emph{Open:} a near-identity averaging transformation for the stochastic McKean--Vlasov loop that keeps the
      frozen-measure correction valid along the moving measure, uniformly in time.
    \item \emph{Done (Theorem J, Section~\ref{sec:thmJ}):} the deterministic case ($\tau=0$, $\sigma=\sigma_0/k_e$)
      at fixed $m$: near every hyperbolic critical point of $J_m$ the real loop has a unique periodic orbit with the
      same stability type, a $k_e$-uniform basin, and the first-order location of Proposition H.7 and Remark H.7a.
    \item \emph{Done (Theorem J$'$, Section~\ref{sec:thmJprime}):} the dithered case at fixed $m$. The invariant law
      of the loop has the Gibbs measure $\exp(-J_m/\tau_0)$ as its parameter marginal up to $O(1/k_e)$, with the
      first-order correction driven by the same drift $b$ as Theorem J; the fast error and the phase are eliminated by
      a Poisson-equation expansion of the generator rather than a transformation of the Fokker--Planck equation.
    \item \emph{Next steps:} uniformity in $m$ --- the mean-field limit of the stationary law (the Gibbs measures of
      $J_m$ converge to the minimiser of the mean-field free energy $F_\tau$ of Theorem H, a propagation-of-chaos
      question for Gibbs measures which is where Conjecture H.6 now lives), the second-order correction, and
      quantitative constants in place of the $O(\varepsilon^2)$.
  \end{itemize}

\item \textbf{Non-reciprocal mean-field dynamics from closed-loop learning (new; Proposition H.7).}
  \begin{itemize}
    \item \emph{Proved at a frozen measure:} the averaged adaptive dynamics is a semi-gradient flow plus a lag term
      (Proposition H.7). Along the real loop a second-order averaging term is added (Remark H.7a).
    \item Its interaction has two antisymmetric parts:
      \begin{itemize}
        \item the performative defect $A_{\mathrm{even}}$ of the state shift, which is present for every $n$ and is
          the only part for $n=1$;
        \item the circulation kernel $K$, an integral over the region that the reference orbit encloses ($n\ge2$).
      \end{itemize}
    \item \emph{Proved (Theorem I, Section~\ref{sec:finitegain}):} reciprocity at order $k_e^{-2}$ is equivalent to
      $A_\mu\equiv0$; for $n=1$ it holds exactly at perfect fit on the orbit, and for $n\ge2$ at perfect fit exactly
      when the orbit carries no circulation and the residual's Jacobian is symmetric along the orbit; no open set of
      measures is reciprocal, for any $n$. The performative defect is bounded by the residual and the antisymmetric
      part of its Jacobian; the circulation defect depends on neither.
    \item \emph{Open questions:}
      \begin{itemize}
        \item Are stationary states unique beyond the perturbative regime?
        \item Can circulation be \emph{designed}, choosing the orbit's orientation and area, to break symmetric traps
          or to speed escape from the collapsed configuration? (Partly answered by Proposition K, below: the
          orientation decides the stability of the collapsed configuration in a window around the critical leak.)
        \item Is there a Lyapunov functional other than $F_\tau$ for the non-reciprocal flow?
      \end{itemize}
    \item \emph{Context:} the closest analogues are performative prediction~\cite{S14}, semi-gradient TD
      learning~\cite{S10}, \cite{S11} and non-reciprocal phase transitions~\cite{S12}.
    \item \emph{Done (Remark H.7a):} deterministic $n=2$ closed-loop runs with the reference traversed both ways
      (same $\rho$). The full first-order theory predicts the $O(1/k_e)$ difference of the learned kernels to 3\% in
      size and 11\% as a vector.
    \item \emph{Done (Proposition K, Section~\ref{sec:thmK}):} at a collapsed configuration the orientation-odd
      $k_e^{-2}$ terms shift the separation Floquet exponents of the real loop by $O(k_e^{-2})$, with a closed-form
      prediction from the averaged data; at the critical leak $\sigma_{\mathrm{crit}}$ the direction of traversal
      decides whether the kernels separate, and the $\sigma_0$ at which it does can be computed in advance.
    \item \emph{Done (Theorem J$'$):} the test with dither. The orientation-odd part of the stationary parameter
      means is $2\varepsilon\int\Theta g_{\mathrm{odd}}\dd\pi$, with $g_{\mathrm{odd}}$ driven by the odd part of
      $b$; measured on the single-kernel instance of Section~\ref{sec:thmJprime} (where, in one state dimension, the
      whole odd effect is $b_{\mathrm{osc}}$).
    \item \emph{New (Proposition J$'$.2): the stationary state is a non-equilibrium steady state.} The first-order
      averaged dynamics carries a stationary current $j_1 = \pi(b - \tau_0G\nabla g)$ exactly when $b$ is not a
      $G$-gradient, which for $m\ge3$ is always the case; its entropy production rate is
      $\varepsilon^2\tau_0^{-1}\pi(\norm{b_\perp}^2_{\Ginv})$; and the real loop's stationary parameter process is
      not reversible. The Lyapunov-functional question for the non-reciprocal flow is therefore a question about a
      non-equilibrium steady state (its free energy is not a Lyapunov functional of the real loop at order
      $\varepsilon$).
    \item \emph{Next steps:} the design of the orbit's area (not only its orientation) as a control input; the
      orientation effect at non-collapsed configurations with several kernels; measuring the current of the real loop
      (below the Monte Carlo resolution in Section~\ref{sec:thmJprime}'s instance; a variance-reduced estimator or a
      stronger instance is needed); the mean-field limit of the current.
  \end{itemize}

\item \textbf{Uniform-in-time propagation of chaos with a plant in the loop.}
  \begin{itemize}
    \item Theorems D and D$'$ are finite-horizon; D$'$ gives the $m^{-1/2}$ rate for the dithered loop by synchronous
      coupling, with constants independent of the temperature and an exponent independent of the dimension for the
      state and the learned function. Uniform-in-time results exist for mean-field Langevin dynamics~\cite{S5} and
      for shallow networks~\cite{R36}.
    \item \emph{Done (Theorem D$''$, Section~\ref{sec:thmDprime}):} uniform in time when the leak dominates the loop
      gain of the adaptive law (a small-gain condition between the plant channel and the particle channel, explicit in
      Lipschitz constants). The certified regime is leak-dominated: the network does not learn (without dither it
      collapses to one kernel; with dither it stays in the cloud of the prior, where the bound has content: the
      coupled distance is of order $m^{-1/2}$ and bounded in time). The result marks where the contraction method
      stops rather than settling the question.
    \item \emph{Open, and now with a negative indication:} the learning regime (kernels separated, leak below the
      critical value of Proposition K), where the plant couples all particles through $e$ and (by H.7) the averaged
      interaction is not reciprocal. The long-horizon run of \code{exp_chaos_uniform.py} ($\sigma = \sigma_K/3$,
      1000\,s) shows the coupled distance growing threefold between the windows $(100,200]$ and $(500,1000]$ while
      keeping the $m^{-1/2}$ law at each time: on that horizon only the finite-horizon Theorem D$'$ is supported.
      Whether the growth saturates (the leak's time scale is $1/\sigma = 2000$\,s), and whether a uniform statement
      holds at all in the learning regime, is the open question; the first draft's claim of empirical uniformity was
      based on a run that was in fact a slow collapse and is withdrawn (review round 6).
    \item \emph{Done (Theorem D$'''$, Section~\ref{sec:thmDppp}):} for the averaged slow loop at $\tau = 0$ the
      question is settled where the fitted configuration is non-degenerately stable: the interaction term of the
      synchronous coupling is a negative square at linear order (flat convexity), the self term carries the
      residual-weighted curvature at the actual residual, and uniform-in-time propagation of chaos at the rate $1/m$
      (mean square) holds under the strict second-order optimality condition along the mean-field flow, with explicit
      constants carrying the transient's $e^{2\Lambda_1}$; the window
      $\kappa_R[\mu_\infty] < \sigma_0 < \sigma_{\mathrm{crit}}$ is where learning and uniformity coexist. The cold
      start of Theorem D$''$'s experiment was outside it on two counts (the expanding transient, and a connected
      support threading the ridges, where the coupling has a floor of pairs in different basins). At finite $m$ the
      stationary coupled distance is the weight-sensitivity law of Proposition D$'''$.5, and Corollary D$'''$.7
      carries the uniform statement to the real loop on a warm start through Theorem J.
    \item \emph{Open:} the measure-weighted form of hypothesis (H) (a bad set of small $\mu_s$-mass, the bootstrap on
      the good particles, the flip fraction $p_m$ and its $m$-law); the transfer to the real loop at finite $k_e$
      through Theorem J$'$'s $O(\varepsilon)$ corrections, uniformly in time; dither (the entropy method
      of~\cite{S5}, whose hypotheses Theorem H verifies, against the coupling's floor).
  \end{itemize}

\item \textbf{Geometry of two-time-scale adaptation (Proposition G(b$'$)).}
  \begin{itemize}
    \item \emph{Proved formally, and checked:} the slow geometry follows the gradient flow of the profile loss in the
      metric induced on the slow manifold, and the slope $S$ measures the departure from plain variable projection.
    \item \emph{Open:} a rigorous long-horizon statement; the $O(1/R)$ corrections, which involve $D^2a^*$ and
      $\partial_\xi M$; gain designs that shape the induced metric.
  \end{itemize}

\item \textbf{Noiseless global convergence.}
  \begin{itemize}
    \item \emph{Done (Proposition K):} the local criterion for the collapsed trap: a collapsed configuration is a
      strict local minimum of the regularised loss iff
      $\sigma_0>\sigma_{\mathrm{crit}}(\bar\vartheta)=\lambda_{\max}(G^{1/2}R_2G^{1/2})$, where $R_2$ is the Hessian
      of the first-variation potential at the atom; below $\sigma_{\mathrm{crit}}$ the noiseless flow leaves the
      trap along the softest separation mode, without dither, and the residual at the atom is what drives the
      separation ($R_2$ is linear in it).
    \item \emph{Open:} a closed-loop separation condition under which the noiseless flow avoids spurious stationary
      points, despite the failure of the Chizat--Bach~\cite{R29} boundary conditions. Candidates are a compactified
      width coordinate and reflecting boundaries (Remark H.3).
  \end{itemize}

\item \textbf{Unbalanced transport, meaning online structure adaptation.}
  \begin{itemize}
    \item Treat the weights as masses (Hellinger--Kantorovich\,/\,Wasserstein--Fisher--Rao) to obtain birth and death
      of kernels with Lyapunov guarantees. The conic particle descent of~\cite{R49} and~\cite{R55} is the static
      analogue.
  \end{itemize}

\item \textbf{Excitation theory in measure space.}
  \begin{itemize}
    \item \emph{Done (Lemma E.0):} a quantitative, uniform UES estimate for the skew-symmetric class, explicit in
      $(k,\alpha,T,\Psi_M,\Psi_D)$, which reproduces the $\omega^{-2}$ law of Remark E.1.
    \item \emph{Done (Lemma E.0$'$):} a strict Lyapunov function whose rate is the averaged rate
      $\lambda_{\min}\langle\Psi^\T\Psi\rangle/k$ up to a factor $1-\delta$, for $k$ above a threshold set by the
      peak factors of $\Psi$, $\dot\Psi$, $\ddot\Psi$ (second-order slaving of the error to its slow manifold and a
      second-order averaging correction, whose mean part vanishes identically); the ill-conditioning of the averaged
      Gramian is paid once, as in the exact rate.
    \item \emph{Done (Corollary E.0$'$.3):} the same Lyapunov function bounds the stationary parameter spread of the
      dithered linearised loop,
      $\limsup\E\norm{\zeta}^2\le(1+O(\varepsilon))k\norm{\Sigma}_F^2/(2(1-\delta)\lambda_{\min})$, within
      $1/(1-\delta)$ of the exact (averaged Langevin) value on isotropic regressors and paying the condition number
      in the trace otherwise: an explicit design formula for how much dither a given tracking budget tolerates.
    \item \emph{Done (Lemma E.0$''$):} the non-periodic case. The period mean is replaced by a forward exponential
      average over a window $T$ (whose derivative identity $T\dot{\bar M}=\bar M-M$ gives the antiderivatives in
      closed form), the peak factors become time-local, and one residual second-order term $\bar N_2$ appears,
      costing $\delta_2=O(1/k)$ in the rate; the guaranteed rate is $(1-\delta-\delta_2)\lambda_T/k$ with
      $\lambda_T$ the exponentially weighted PE level of the window. Checked on a quasi-periodic regressor and on a
      regressor driven by a Lorenz orbit, where no periodic construction applies.
    \item \emph{Open:} the continuum form of (PE); the trajectory kernel
      \[K_T(\vartheta,\vartheta')=\int\Phi(x(s);\vartheta)\Phi(x(s);\vartheta')\dd s;\]
      identifiability of measures up to symmetry; the relation to~\cite{R16}. Remark E.1 shows that the PE level
      alone does not predict the rate. For Lemmas E.0$'$ and E.0$''$: lowering the threshold $k_0$ (a third-order
      slaving trades $\rho_{DD}$ for the peak factor of $\dddot\Psi$ one order higher); the choice of the window $T$
      as an optimisation; the nonlinear Theorem E with the constants of E.0$''$ (Corollary E.0$'$.2's steps carry over
      verbatim, with $\lambda_T$ in place of $\lambda_{\min}(\bar M)$).
  \end{itemize}

\item \textbf{Transport-regularised adaptive laws.}
  \begin{itemize}
    \item Wasserstein proximal (JKO) discrete-time adaptive laws.
    \item \emph{Done (Theorem C$'$, Section~\ref{sec:thmCprime}):} the OT leak toward priors with a different number
      of atoms, using optimal plans with barycentric projection; it is Lloyd's algorithm in continuous time, and its
      bias is the centroid defect of the ideal.
    \item \emph{Open (in part):} its mean-field form --- done for an atomic prior (the Laguerre-cell leak,
      Proposition D$'''$.12), open for a general prior $\nu$ and a general $\mu$ (where the barycentric projection
      is the superdifferential of $\frac12W^2(\cdot,\nu)$; Ambrosio--Gigli--Savar\'e~\cite[Thm.~10.4.12]{ags2008})
      --- and adaptive priors ($\nu$ itself learned).
    \item \emph{Done (Proposition D$'''$.9, Section~\ref{sec:thmDppp}):} the nearest-atom leak toward a prior
      measure inherits Theorem D$'''$, Proposition D$'''$.5 and Corollary D$'''$.7 inside the Voronoi cells; the
      prior is a design input of the window (maximal margin $\kappa = \sigma_0$ at a matching prior, the window
      closing as the prior is misplaced) and of the floors (which grow with the stationary amplitudes).
    \item \emph{Done (Proposition D$'''$.12):} the barycentric OT leak's mass constraint against the weight
      fluctuation. The nearest-atom leak conserves the cluster masses, and the finite-$m$ floor of
      Proposition D$'''$.5 is the memory of the initial cloud that this conservation carries; the OT leak constrains
      the masses to the prior's, so on the sizes commensurate with the masses the finite-$m$ stationary measure is
      the mean-field one exactly (the replicated weighted configuration is a critical point for every such $m$, by
      total unimodularity and $c$-cyclical monotonicity), and otherwise differs from it by at most $K_0 - 1$ split
      particles; for an atomic prior the mean-field form of the barycentric leak is the Laguerre-cell leak.
      \emph{Open:} the uniform-in-time coupling through the transit (the $m$-system is no longer an i.i.d.\ sample
      after the mass transfer); a soft (entropic or penalised) mass constraint interpolating between the two leaks;
      the mean-field form for a general prior $\nu$ (the barycentric projection as the superdifferential of
      $\frac12W^2(\cdot,\nu)$, Ambrosio--Gigli--Savar\'e~\cite[Thm.~10.4.12]{ags2008}) and adaptive priors ($\nu$,
      and its masses, learned).
    \item Transfer: transporting a learned measure to a new plant.
  \end{itemize}

\item \textbf{Quantisation and the curse of dimensionality (Corollary B.6).}
  \begin{itemize}
    \item \emph{Done (Theorem C$'$):} an adaptive law whose leak tracks the equal-mass centroidal quantisers of a prior
      measure (centroidal Voronoi tessellations~\cite{S20}; coverage control~\cite{S18}, \cite{R89}), and Theorem D$'$'s
      observation that the state and the learned function converge at the rate $m^{-1/2}$, whose exponent is
      dimension-free: the curse of dimensionality of Corollary B.6 lives in the $W$-distance of the parameter
      measures, which the loop never evaluates.
    \item \emph{Open:} second-order (centroid) quantisation rates for the ideal measure on the lifted space;
      approximation rates that do not go through $W$ (Maurey-type).
  \end{itemize}

\item \textbf{Symmetric systems beyond kernels.} The OT quotient applies verbatim to swarms of identical agents,
  mixtures and multi-head models (\cite{R88} and~\cite{R89} are the control-side precedents).

\item \textbf{Deep approximators.} Hierarchical (multi-layer) mean-field measures, with a layer-wise OT Lyapunov
  function.

\end{enumerate}
